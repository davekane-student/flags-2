Flag Viewer

viewof selected = Inputs.select(codes, {
  label: "Select a flag:",
  value: "us"
})

// Helper to render a flag card with its 2-letter code
renderFlag = (code, width = "180px") => htl.html`
  <div style="text-align: center;">
    <img src="flags/${code}.png" style="width: ${width}; border: 1px solid #ccc; display: block;">
    <div style="margin-top: 0.25rem; font-weight: bold;">${code}</div>
  </div>
`

htl.html`
  <div style="margin-top: 1rem;">
    <h3>Selected Flag</h3>
    ${renderFlag(selected, "300px")}

    <h3 style="margin-top: 1.5rem;">3 Most Similar Flags</h3>
    <div style="display: flex; gap: 1rem; flex-wrap: wrap;">
      ${matches[selected].similar.map(c => renderFlag(c))}
    </div>

    <h3 style="margin-top: 1.5rem;">3 Most Different Flags</h3>
    <div style="display: flex; gap: 1rem; flex-wrap: wrap;">
      ${matches[selected].different.map(c => renderFlag(c))}
    </div>
  </div>
`
